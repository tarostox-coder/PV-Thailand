1. Dynamic Column Handling ใน Power Query (M Language)

ใช้ MissingField.Ignore + Table.ColumnNames() แทนการ hardcode ชื่อ Column

// ① Auto-find Header Row (ไม่ต้องรู้ว่า header อยู่แถวไหน)
let
  Src    = Excel.Workbook(File.Contents(FilePath), null, true),
  Sheet  = Src{[Item="Project Data",Kind="Sheet"]}[Data],
  HdrIdx = List.PositionOf(
    Table.Column(Sheet,"Column1"),
    List.First(List.Select(
      Table.Column(Sheet,"Column1"),
      each Text.Contains(Text.From(_??""),"ชื่อโครงการ")
    ))
  ),
  Promoted = Table.PromoteHeaders(Table.Skip(Sheet, HdrIdx),
               [PromoteAllScalars=true]),

  // ② Keep ALL columns — never drop unknowns
  AllCols = Table.ColumnNames(Promoted),

  // ③ Safe rename — MissingField.Ignore = ไม่ error ถ้าไม่มี column
  SafeRename = Table.RenameColumns(Promoted,
    List.Select(
      {{"ขนาด (kWp)", "Capacity_kWp"},
       {"สถานะปัจจุบัน","Status"},
       {"ประเภทสัญญา","ContractType"},
       {"Start Date","StartDate"},
       {"End Date","EndDate"},
       {"% Progress","Progress"}},
      each List.Contains(AllCols, _{0})  // only rename if exists
    )
  ),

  // ④ Tag all extra columns into one JSON-like string
  CoreFields = {"NO","ชื่อโครงการ","Capacity_kWp","ContractType",
                "สถานที่","Lead Sale","วิศวกร","Status",
                "StartDate","EndDate","Progress"},
  ExtraFields = List.Difference(Table.ColumnNames(SafeRename), CoreFields),
  WithExtra   = Table.AddColumn(SafeRename, "_ExtraCols",
    each Text.Combine(
      List.Transform(ExtraFields, col =>
        col & "=" & Text.From(Record.Field(_,col)??"")
      ), " | "
    )
  ),

  // ⑤ Clean up types — use try/otherwise for safety
  Typed = Table.TransformColumns(WithExtra, {
    {"Capacity_kWp", each try Number.From(_) otherwise null, type number},
    {"Progress",     each try Number.From(_) otherwise 0,    type number},
    {"StartDate",    each try Date.From(_)   otherwise null, type date},
    {"EndDate",      each try Date.From(_)   otherwise null, type date}
  }, MissingField.Ignore)  // ← ถ้าไม่มี column ก็ไม่ error
in
  Typed

2. DAX Measures — Safe & Future-proof

// ── KPI Measures ───────────────────────────────────────────────────────
Total_Projects   = COUNTROWS('Projects')

Win =
  CALCULATE([Total_Projects],
    FILTER('Projects', LOWER('Projects'[Status]) IN {"win","ดำเนินโครงการ","เซ็นสัญญาแล้ว"})
  )

In_Progress =
  CALCULATE([Total_Projects],
    NOT LOWER('Projects'[Status]) IN {"win","ดำเนินโครงการ","เซ็นสัญญาแล้ว","lost","hold"}
  )

// ── Capacity (IFERROR ป้องกัน null/blank) ─────────────────────────────
Total_kWp   = SUMX('Projects', IFERROR('Projects'[Capacity_kWp], 0))
Win_kWp     = CALCULATE([Total_kWp], LOWER('Projects'[Status]) IN {"win","ดำเนินโครงการ","เซ็นสัญญาแล้ว"})
Win_Rate    = DIVIDE([Win_kWp], [Total_kWp], 0)
Avg_kWp     = AVERAGEX(FILTER('Projects','Projects'[Capacity_kWp]>0), 'Projects'[Capacity_kWp])

// ── Alert Status (รองรับ null EndDate) ───────────────────────────────
Alert_Status =
  VAR DL  = IFERROR(DATEDIFF(TODAY(), MAX('Projects'[EndDate]), DAY), BLANK())
  VAR Cat = LOWER(MAX('Projects'[Status]))
  RETURN
    SWITCH(TRUE(),
      Cat = "lost",               "Closed",
      Cat = "hold",               "On Hold",
      ISBLANK(DL),                "No Date",
      DL < 0,                     "⚠️ Overdue",
      DL <= 30,                 "🔴 Near Deadline",
      DL <= 90,                 "🟡 Approaching",
                                  "✅ On Track"
    )

// ── Dynamic Measure (Field Parameter) ────────────────────────────────
// สร้างใน Desktop: Modeling → New Parameter → Fields
// เลือก: [Total_kWp], [Win_kWp], [Total_Projects], [Avg_kWp]
// ผลคือ User เลือก Metric ที่ต้องการดูใน Visual ได้เองแบบ Dynamic

3. Schema-Agnostic Visual Design

✅ แนะนำ (DO)
• Field Parameters ให้ User เลือก Metric/Axis เอง
• Matrix Visual แทน Table แบบ hardcode
• Calculation Groups สำหรับ format rules
• MissingField.Ignore ทุก rename/transform
• IFERROR / ISBLANK ใน DAX ทุกจุด
• เก็บ Extra Cols เป็น JSON string ใน 1 Column
• ใช้ Bookmark + Button แทน hardcode filter
❌ หลีกเลี่ยง (AVOID)
• Hardcode column name ใน DAX โดยตรง
• ใช้ Column Index แทน Column Name
• Remove Columns ใน Power Query
• Table Visual ที่ผูก Column ทุกตัวแบบตายตัว
• DAX ที่ไม่มี Error Handling
• Snapshot/static data ใน Dashboard
• ใช้ Paginated Report สำหรับ Live Filter

1บันทึกไฟล์ Excel ใน OneDrive for Business / SharePoint (ไม่ใช่ Local) — ให้หลายคนแก้ไขพร้อมกันได้

3Power BI Service → Dataset Settings → Scheduled Refresh ทุก 1 ชม. (Pro) หรือทุก 30 นาที (Premium)
4Power Automate: "When a file is modified (SharePoint)" → Action: "Refresh a Power BI dataset" → Near Real-time ทันทีที่บันทึก
5เพิ่ม DAX: LastRefresh = "🔄 " & FORMAT(NOW(),"DD/MM/YYYY HH:MM น.") แสดงบน Dashboard
6ใช้ Incremental Refresh (Premium) ถ้าข้อมูลเยอะ — refresh เฉพาะแถวใหม่ ไม่ reload ทั้งหมด
